# LC 4006
# https://leetcode.com/problems/count-valid-prefixes/


->

Return the `number` of `valid `prefixes`` of s.


prefix is valid if:
    -  its characters can be 
        rearranged to form 
        an `alternating` string.



alternating:
    - if `NO` two adjacent characters are equal.



idea 1) looping + `alternating` check
  -> check if 0, 1
    - abs(cnt(1) - cnt(0)) >= 1
    - prefix len <= 2




----------------------


ex 1)

nput: s = "00101"

Output: 3



->


00101




00101  ->  01010


001010

0101



----------------------


import math

class Solution(object):
    def countValidPrefixes(self, s):
        """
        :type s: str
        :rtype: int
        """
        # edge
        if len(s) == 1:
            return 1
        if len(s) == 2:
            if s[0] == s[1]:
                return 1
            return 2


        cnt = 0
        z_cnt = 0
        o_cnt = 0

        for i in range(len(s)):
            val = s[i]
            # ???
            # update cnt first ???
            if val == 0:
                z_cnt += 1
            else:
                o_cnt += 1

            if i == 0:
                cnt += 1
            if i == 1:
                if z_cnt == o_cnt:
                    cnt += 1

            if math.abs(o_cnt - z_cnt) <= 1:
                cnt += 1


        return cnt

from collections import Counter


class Solution(object):
    def maximumWidth(self, planks):
        """
        :type planks: List[int]
        :rtype: int
        """
        # edge
        target = planks[0]
        
        c_map = Counter(planks)
        max_cnt = max(c_map.valus())
        for k in c_map:
            # ??? is it possible
            # to k has same cnt ???
            if c_map[k] == max_cnt:
                target = k
                break

        # {prefix: cnt}
        # prefix_b - prefix_a == k
        # -> prefix_a = prefix_b - k


        cnt = 0
        
        p_map = {}
        prefix = 0

        for x in planks:
            if prefix - target in p_map:
                cnt += 1


            prefix += x

            # ?? update map
            p_map[prefix] = p_map.get(prefix, 0) + 1

# LC 4008
# https://leetcode.com/problems/minimum-initial-strength-to-defeat-all-monsters/


->

Return the `minimum` `initial` `strength `
required to defeat `ALL` monsters.


start with a x strength (x > 0)
and fight with monster from left to right





monsters[i]: str of i monster

boosts[i] = [li, ri, vi]
    - li: start idx
    - ri: end idx
    - vi: `emporary bonus`


boost may `overlap`
    -> e.g. accumulated adding on



can defeat the monster ONLY
    - strength + bonus >= monster[i]


- After defeating the monster, 
    only your current strength `decreases` by monsters[i]. 
    If it becomes negative, it is set to 0.




Note: The temporary bonus is used 
    ONLY to `determine` whether the 
    current monster can be defeated. 
    -> It does NOT otherwise `change your current strength.`



# idea 1) simulation (brute force)


# idea 2) binary search

class Solution(object):
    def minInitialStrength(self, monsters, boosts):
        # edge

        l = min(monsters) 
        r = max(monsters)



        n = len(monsters)

        prefix_list = [0] * (n+1)
        for s, e, val in boosts:
            prefix_list[s] += val

            # ??
            if e + 1 < n:
                prefix_list[e] -= val



        # ??
        min_str = r

        while r >= l:
            mid = l + (r - l) // 2
            # ???
            if self.can_defeat(monsters, boosts, mid, prefix_list)
                min_str = mid
                # keep finding smaller candidates
                r = mid - 1
            else:
                l = mid + 1

        return min_str


    def can_defeat(self, monsters, boosts, x, prefix_list):
        # edge
        n = len(monsters)

        for i in range(n):
            if x < 0:
                x = 0
            val += (x + prefix_list[i])
            if val < monsters[i]:
                return False

            x = x - monsters[i]
            if x < 0:
                x = 0

        return True
