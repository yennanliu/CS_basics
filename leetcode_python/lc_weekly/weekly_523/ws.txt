class Solution(object):
    def maxPrimes(self, n, s):
        """
        :type n: int
        :type s: int
        :rtype: List[int]
        """
        # edge

        self.prime_set = set()

        self.get_primes(n, s)
        self.primes = list(self.prime_set)


        res = []
        #tmp = []
        _sum = 0

        l = 0

        for r in range(len(self.primes)):

            val_r = self.primes[r]
            _sum += val_r

            while _sum > s:
                val_l = self.primes[l]
                _sum -= val_l
                l += 1


            if r - l + 1 > len(res):
                res = self.primes[l, r]


        return res
