class Solution(object):
    def isPalindromic(self, s):
        """
        :type s: str
        :rtype: bool
        """

        # Convert every character to its 8-bit ASCII representation
        tmp = ""

        for x in s:
            tmp += self.helper(x)

        # Check whether the binary string is a palindrome
        l = 0
        r = len(tmp) - 1

        while l < r:
            if tmp[l] != tmp[r]:
                return False

            l += 1
            r -= 1

        return True

    def helper(self, x):
        # Get ASCII value
        ascii_val = ord(x)

        # Convert to exactly 8-bit binary
        # Example: 102 -> "01100110"
        return format(ascii_val, "08b")

class Solution(object):
    def findDisappearedNumbers(self, nums, lower, upper):
        """
        :type nums: List[int]
        :type lower: int
        :type upper: int
        :rtype: List[List[int]]
        """

        # Only keep numbers inside [lower, upper]
        nums = sorted(set(
            x for x in nums
            if lower <= x <= upper
        ))

        res = []

        # Current smallest number we expect
        prev = lower - 1

        for x in nums:

            # There is a missing range between prev and x
            if x > prev + 1:
                res.append([prev + 1, x - 1])

            prev = x

        # Check the remaining range after the last number
        if prev < upper:
            res.append([prev + 1, upper])

        return res

from collections import Counter
import math


class Solution(object):
    def longestSubarray(self, nums, k):
        """
        :type nums: List[int]
        :type k: int
        :rtype: int
        """

        # factor[i] = prime factors of nums[i]
        factor = []

        for x in nums:
            #factor.append(self.get_prime_factor(x))
            factor.append(self.get_prime_factor_v2(x))

        # Count how many numbers in the current window
        # contain each prime factor.
        factor_cnt = Counter()

        # Number of distinct prime factors in window
        distinct = 0

        max_len = 0
        l = 0

        for r in range(len(nums)):

            # Add nums[r]'s prime factors
            for p in factor[r]:
                if factor_cnt[p] == 0:
                    distinct += 1

                factor_cnt[p] += 1

            # Shrink window if too many distinct factors
            while distinct > k:

                for p in factor[l]:
                    factor_cnt[p] -= 1

                    if factor_cnt[p] == 0:
                        distinct -= 1

                l += 1

            # Current window [l, r] is valid
            max_len = max(max_len, r - l + 1)

        return max_len


    def get_prime_factor_v2(self, x):
            """Returns the set of distinct prime factors of x."""
            factors = set()
            d = 2
            while d * d <= x:
                if x % d == 0:
                    factors.add(d)
                    while x % d == 0:
                        x //= d
                d += 1
            if x > 1:
                factors.add(x)
            return factors

    def get_prime_factor(self, x):
        """
        Return the set of distinct prime factors of x.

        Example:
            12 -> {2, 3}
            18 -> {2, 3}
            25 -> {5}
            1  -> {}
        """

        res = set()

        # Handle factor 2
        while x % 2 == 0:
            res.add(2)
            x //= 2

        # Handle odd factors
        p = 3

        while p * p <= x:
            while x % p == 0:
                res.add(p)
                x //= p

            p += 2

        # Remaining x is prime
        if x > 1:
            res.add(x)

        return res
