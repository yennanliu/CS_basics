# LC 4038
# https://leetcode.com/problems/count-integers-appearing-in-a-single-block/


->

Return the `number` of `distinct` 
`special integers` in nums.

 
nums: int arr


x is special:
	-  special if `all occurrences`  of x in 
	   nums appear in a `single contiguous block.`


# idea 1) brute force

# idea 2) hashmap
	
	map: {val: [idx_1, idx_2, ...]}

	-> check how many of val is `special`


from collections import defaultdict

class Solution(object):
    def countSpecialIntegers(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # edge
        if not nums or len(nums) == 0:
        	return 0
        
        if len(nums) == 1:
        	return 1

        # map: {val: [idx_1, idx_2, ...]}
        my_map = defaultdict(list)

        for i in len(nums):
        	val = nums[i]
        	my_map[val].append(i)

        cnt = 0

        for k in my_map:
        	is_perfect = True
        	for i in range(1, len(my_map[k])):
        		if my_map[k][i] != my_map[k][i-1] + 1:
        			is_perfect = False
        			break

        	if is_perfect:
        		cnt += 1

       return cnt

# LC 4039
# https://leetcode.com/problems/sum-of-decoded-numbers/description/



->

Return the `sum` of the `decoded values` 
of all `elements in nums`, modulo 109 + 7.


nums: int arr


nums[i] 
	is an encoded integer
	xi, yi


to decode nums[i] :

	- widthi = nums[i] % 10

	- di = floor(nums[i] / 10)


	- xi as the integer formed by the `first widthi`
		 digits of the decimal representation of di.


	- yi as the integer formed by `all remaining `
	 digits of the decimal representation of di.


The decoded value of nums[i] is (xi) ^ (yi)


# idea 1) math


class Solution(object):
    def sumDecoded(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # edge
        if not nums or len(nums) == 0:
        	return 0

        ans = 0

        for x in nums:
        	width, d = self.helper(x)

        	# ??
        	x = int(str(d[:width]))
        	y = int(str(d[width + 1:]))

        	ans += (x)^ y

       return ans


    # given x, return width, d
    def helper(self, x):
		#widthi = x % 10
		#di = floor(x / 10)

		return x % 10, floor(x / 10)
