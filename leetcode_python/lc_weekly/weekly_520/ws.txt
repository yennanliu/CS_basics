Return the `number` of `pairs` of `indices` (i, j)
 such that 0 <= i < j < n 
 and 
 intervals[i] and intervals[j] intersect 相交.


intersect:
    Two intervals intersect if they have at least one point in common, 
    including when they only share an endpoint.




intervals: 2D arr, n elements


intervals[i] = [starti, endi] 
    - close intervals

# idea 2) brute force (double loop)

ex 1)

Input: intervals = [[1,2],[2,3],[3,4]]

Output: 2


->


 |----|
 1    2

      | ----- |
      2       3
            
              |---|
              3   4



ex 2)


Input: intervals = [[1,5],[2,4],[3,6]]

Output: 3



 |-------------|

 1             5
    |------| 
    2      4

      |-----------|
      3           6

#  idea 2) brute force (double loop)

class Solution(object):
    def countIntersectingIntervals(self, intervals):
        """
        :type intervals: List[List[int]]
        :rtype: int
        """
        # edge
        if not intervals or len(intervals) == 0:
            return 0

        if len(intervals) == 1:
            return 0

        # sort: on 1st: small -> big, second: small -> big,
        # ???
        intervals.sort(key = lambda x: (x[0], x[1]) )

        cnt = 0

        n = len(intervals)

        for i in range(n):
            for j in range(i+1, n):
                # overlap
                """
                since already sort on 1st, 2nd element

                |-------|
                           |----|
                """ 
                cur = intervals[j]
                prev = intervals[i]

                if not (cur[0] > prev[1]):
                    cnt += 1
                
                # optimization:
                # if NOT possible to overlap anymore,
                # negelect all of the remain loop for j
                else:
                    break


        return cnt

class Solution(object):
    def countIntersectingIntervals(self, intervals):
        """
        :type intervals: List[List[int]]
        :rtype: int
        """
        # edge
        if not intervals or len(intervals) == 0:
            return 0

        if len(intervals) == 1:
            return 0


        tmp = []

        for s, e in intervals:
            tmp.append([s, 1])
            tmp.append([e, -1])


        # sort ??
        # intervals.sort(key = lambda x: (x[0], x[1]) )
        tmp.sort(key = lambda: x[0])


        max_cnt = 0
        tmp_cnt = 0


        for _time, _status in tmp:
            if _status == 1:
                tmp_cnt += 1

            else:
                tmp_cnt -= 1

            # ???
            max_cnt = max(max_cnt, self.helper(tmp_cnt))


        return max_cnt


    def helper(self, x):
        if x <= 1:
            return x

        """
        C(x, 2)

        = (x!) * (2!) / (x-2) !

        = (x-1)(x) / 2
        """
        return (x-1)(x) // 2
