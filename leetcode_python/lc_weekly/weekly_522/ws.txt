# LC 4070
# https://leetcode.com/problems/minimum-rotations-to-dial-a-number-i/description/
#------------------------------------------------


->
Return the `minimum total number` of 
`rotations` needed to `dial every digit of s.`

 

#----------------------


ex 1)


Input: s = "0192837465"

Output: 25


0123456789






#----------------------


# idea 1) brute force + min(left_move, right_move)


class Solution(object):
    def minRotations(self, s):
        """
        :type s: str
        :rtype: int
        """
        # edge
        if not s or len(s) == 0:
            return 0

        if len(s) == 0:
            val = int(s)
            return min(abs(9 - val) + 1, int(val))

        moves = 0
        pos = 0

        for x in s:

            val = int(s)

            """
            0123456789
               i    j

            """

            move = min(abs(val - pos), (pos + 1) + (9 - int))

            moves += move

            pos = int(x)


        return moves

# LC 4071
# https://leetcode.com/problems/minimum-rotations-to-dial-a-number-ii/description/
#------------------------------------------------



->
Return the minimum total number of `rotations` 
needed to dial the string 

after optimally choosing whether to perform the 
op and which suffix to reverse.


NOTE:

    Before dialing, 
    you may perform the following operation `at most once:`

        Choose an index k such that 0 <= k < n 
        and reverse the suffix s[k..n - 1].





# idea 1) brute force:
    
    try op on every idx (also try NOT do the op),
    and get the min rotation as res

class Solution(object):
    def minRotations(self, n, s):
        """
        :type n: int
        :type s: str
        :rtype: int
        """
        # edge
        if not s or len(s) == 0:
            return 0

        min_rotate = self.helper(s)
        n = len(s)

        for i in range(n):
            tmp_s = s[0: i] + s[i+1:][::-1]

            min_rotate = min(min_rotate, self.helper(tmp_s))


        return min_rotate


     def helper(self, s):
        op = 0
        pos = 0

        for x in s:
            
            val = int(x)

            move = min(abs(val - pos), (pos + 1) + (9 - int))

            op += move

            pos = val


        return op

# LC 4072
# https://leetcode.com/problems/maximum-alternating-subarray-sum-with-one-deletion/description/
#------------------------------------------------



->

Return the `maximum` possible
 `alternating sum` of the `chosen subarray.`



nums: int arr


Can delete `at most one ` element from nums


subarray:
    is a `contiguous` non-empty 
    sequence of elements within an array.




# idea 1) greedy:   
        try every possible op, and try every sub arr


# idea 2) greedy ???
    
    2 cases

        - NOT delete, find the max sub arr sum
        - delete, find the max sub arr sum


# time: O(N) + O(N * (N + N)) ~= O(N ^ 2)
# space:  O(N)

class Solution(object):
    def maxAlternatingSum(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # edge
        if not nums or len(nums) == 0:
            return 0
        if len(nums) == 1:
            return nums[0]

        n = len(nums)

        max_sub = self.get_max_sub(nums)

        for i in range(n):
            # time complexity on below ???
            _nums = nums[0:i] + nums[i+1: ]

            max_sub = max(max_sub, self.get_max_sub(_nums))


        return max_sub



    # slide window ???
    # 1D DP: kanban
    """
    nums = [1,-2,1,2,-4]


    """
    def get_max_sub(self, nums):
        #res = 0
        n = len(nums)

        local_max = nums[0]
        local_min = nums[0]

        global_max = nums[0]

        for r in range(1, n):

            val = nums[r]

            local_max += val
            local_min += val

            global_max = max(val, local_max, local_min)


            if local_max < 0:
                local_max = 0

        return global_max
