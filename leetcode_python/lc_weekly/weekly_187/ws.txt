# LC 3992
# https://leetcode.com/problems/rearrange-string-to-avoid-character-pair/


->

Return `any valid` string t.


s: string
x, y: lower case eng letter

re-arrange s to create a new str t
such that  


1. t is `permutation  排列` of s
2. every `y` exists before `x` in t





# idea 1) hashmap + str op

class Solution(object):
    def rearrangeString(self, s, x, y):
        # edge
        if x not in s:
            return s

        # {val : cnt}
        cnt_map = {}
        for val in s:
            cnt_map[val] = cnt_map.get(val, 0) + 1

        res = ""
        res += (cnt_map.get(y) * y)
        del cnt_map[y]

        for k in cnt_map.keys():
            res += (cnt_map.get(k) * k)


        return res



# LC 3993
# https://leetcode.com/problems/maximum-value-of-an-alternating-sequence/description/


->

Return the `maximum` possible element 
that can appear in `any valid sequence.`

 

int: n, s, m

seq is valid (len = n) if:
    - sqe[0] = s
    - seq[0] > seq[1] < seq[2] > ...,
       or
    - seq[0] < seq[1] > seq[2] < ....


    - For every adjacent pair, 
      `|seq[i] - seq[i - 1]| <= m`



# idea 1) greedy + 2 cases

class Solution(object):
    def maximumValue(self, n, s, m):
        # edge

        max_val = s

        tmp_1 = [s]
        tmp_2 = [s]

        # case 1) seq[0] < seq[1] > seq[2] < ....
        for i in range(1, s):
            # ??
            if i % 2 == 1:
                #_next = tmp_1[0] - 1
                tmp_1[i] = tmp_1[i-1] + m
            else:
                tmp_1[i] = tmp_1[i-1] - m



        # case 2) seq[0] > seq[1] < seq[2] > ...,
        for i in range(1, s):
            # ??
            if i % 2 == 0:
                #_next = tmp_1[0] - 1
                tmp_2[i] = tmp_2[i-1] + m
            else:
                tmp_2[i] = tmp_2[i-1] - m


        return max(max_val, max(tmp_1), max(tmp_2))
