# LC 3992
# https://leetcode.com/problems/rearrange-string-to-avoid-character-pair/


->

Return `any valid` string t.


s: string
x, y: lower case eng letter

re-arrange s to create a new str t
such that  


1. t is `permutation  排列` of s
2. every `y` exists before `x` in t





# idea 1) hashmap + str op

class Solution(object):
    def rearrangeString(self, s, x, y):
        # edge
        if x not in s:
            return s

        # {val : cnt}
        cnt_map = {}
        for val in s:
            cnt_map[val] = cnt_map.get(val, 0) + 1

        res = ""
        res += (cnt_map.get(y) * y)
        del cnt_map[y]

        for k in cnt_map.keys():
            res += (cnt_map.get(k) * k)


        return res

# brute force

class Solution(object):
    def minAdjacentSwaps(self, nums, a, b):
        # edge
        if is_valid(nums):
            return 0

        cnt = 0
        n = len(nums)

        # ??
        for i in range(n-1, -1, 0):
            for j in range(i+1, -1, -1):
                # ???
                if is_valid(nums):
                    return cnt % ((10**9) + 7)

                if nums[i] > nums[j]:
                    # swap
                    nums[i], nums[j] = nums[j], nums[i]
                    cnt += 1

        return cnt % ((10**9) + 7)



    def is_valid(nums, a, b):
        n = len(nums)
        group = 0
        for i in range(n):
            val = nums[i]
            if group == 0:
                if not val < a:
                    return False
            elif group == 1:
                if not (a <= val <= b):
                    return False
            else:
                if not val > b:
                    return False

        return True
