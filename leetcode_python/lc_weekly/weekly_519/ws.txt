# LC 4052
# https://leetcode.com/problems/cyclically-shift-rows-and-columns/description/
#------------------------------------------------


->

Return the `resulting gri`d after 
performing `all the shifts.`



2D matrix: n x n


rowShift, colShift: int arr, size = n




rowShift[i]: 
    the number of positions to 
    cyclically shift the i-th row of grid to the `left.`



colShift[j]:
    represents the number of positions to cyclically 
    shift the j-th column of grid `upward.`




# idea 1) brute force (simulate the process)


class Solution(object):
    def cyclicShift(self, n, grid, rowShift, colShift):
        if not grid or len(grid) == 0:
            return None

        if len(grid) == 1 and len(grid[0]) == 1:
            return grid

        """
        A cyclic left shift of a row by k positions moves the element at column j to column (j - k + n) % n. All other rows remain unchanged.

        """
        for y in rowShift:
            # ???
            for j in range(n):
                tmp_val = grid[y][x]
                grid[y][(j - y + n) % n] = tmp_val

        for x in colShift:
            # ???
            for i in range(n):
                tmp_val = grid[y][x]
                grid[(i - x + n) % n][x] = tmp_val

        return grid

# LC 4053
# https://leetcode.com/problems/minimum-operations-to-make-every-element-palindromic/description/
#------------------------------------------------


->

Return the `minimum number of operations` 
required to make every element in nums a `positive` palindrome 回文.




nums: int arr



# idea 1) brute force


# idea 2) greedy
    - if len == 2:
        -> min move to len == 1

    - else:
        -> find the bigger, and smaller next `positive` palindrome

        -> e.g. : 125

        -> smaller: 121 or 2 ...

        -> bigger: 525 or 626 or 727 .....



class Solution(object):
    def minOperations(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # edge
        if not nums or len(nums) == 0:
            return 0

        cnt = 0

        for x in nums:
            if not self.is_palindrome(x):
                cnt += self.get_op(x)

        return cnt


    def get_op(self, x):

        # len == 2
        if len(str(x)) == 2:
            # ??
            return  floor((x - 9) / 2)

        incre_op = 0
        decre_op = 0

        tmp = x

        # try smaller
        while not self.is_palindrome(str(tmp)):
            tmp -= 2


        tmp = x

        # try bigger
        while not self.is_palindrome(str(tmp)):
            tmp += 2

        return min(incre_op, decre_op)


    def is_palindrome(self, x):
        if len(x) <= 1:
            return True

        x_str = str(x)
        l = 0
        r = len(x_str) - 1

        while r > l:
            if x[l] != x[r]:
                return False
            l += 1
            r -= 1

        return True

# LC 4054
# https://leetcode.com/problems/count-shadow-pairs-i/description/
#------------------------------------------------


->

Return the `total number` of `shadow` pairs.`




nums: int arr, size = n


(i, j) is `shadow pair`:
    if ALL of below are true:

        - 0 <= i < j < n

        - nums[i] < nums[j]

        - DOES NOT exist an idx k
          s.t.

            - i < k < j 
            and 
            - nums[k] < nums[i] < nums[j].





# idea 1) brute force
    - double loop
        loop over nums
        and check above `shadow pair` condition


# idea 2) 2 pointers + smallest_till_now, biggest_till_now

class Solution(object):
    def shadowPairs(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        # edge
        if not nums or len(nums) <= 1:
            return 0

        if len(nums) == 2:
            return 1 if nums[1] > nums[0] else 0

        cnt = 0

        smallest_seen = []
        biggest_seen = []

        smallest = float('inf') # ??
        biggest = float('-1 * inf') # ??

        for x in nums:
            smallest = min(x, smallest)
            smallest_seen.append(x)


        for x in nums:
            biggest = max(x, biggest)
            biggest.append(x)


        # slide window
        n = len(nums)

        for l in range(n - 1):
            r = l + 1

            # ??
            while r < n:

                l_val = nums[l]
                r_val = nums[r]

                if r_val > l_val and r - l == 1:
                    cnt += 1

                # neglect the remaining loop (with cur r) 
                if r - l > 1 and nums[r - 1] > l_val:
                    break

                if r_val > l_val and smallest_seen[r] >= l_val and biggest_seen[r] >= l_val:
                    cnt += 1


        return cnt
